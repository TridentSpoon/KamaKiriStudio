cmake_minimum_required(VERSION 3.24)
project(KamaKiriStudio VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus Network Test QuickWidgets)
find_package(KF6Config REQUIRED)
find_package(KF6IdleTime REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(IDLE_PROTOCOL ${CMAKE_CURRENT_SOURCE_DIR}/protocols/ext-idle-notify-v1.xml)
add_custom_command(OUTPUT idle-client.h COMMAND ${WAYLAND_SCANNER} client-header ${IDLE_PROTOCOL} ${CMAKE_CURRENT_BINARY_DIR}/idle-client.h DEPENDS ${IDLE_PROTOCOL})
add_custom_command(OUTPUT idle-protocol.c COMMAND ${WAYLAND_SCANNER} private-code ${IDLE_PROTOCOL} ${CMAKE_CURRENT_BINARY_DIR}/idle-protocol.c DEPENDS ${IDLE_PROTOCOL})
add_library(studio_core src/core.cpp src/looks.cpp src/palettes.cpp src/desktop.cpp src/input_monitor.cpp ${CMAKE_CURRENT_BINARY_DIR}/idle-protocol.c ${CMAKE_CURRENT_BINARY_DIR}/idle-client.h)
target_include_directories(studio_core PUBLIC src)
target_include_directories(studio_core PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(studio_core PUBLIC Qt6::Widgets Qt6::DBus KF6::ConfigCore KF6::IdleTime PkgConfig::WAYLAND)
target_compile_definitions(studio_core PRIVATE STUDIO_ASSET_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_compile_options(studio_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kamakiri-studio src/main.cpp src/window.cpp src/desktop_window.cpp src/about.cpp src/wallpaper_gallery.cpp src/popup.cpp assets/start-menu.qrc)
target_link_libraries(kamakiri-studio PRIVATE studio_core Qt6::Network Qt6::QuickWidgets KF6::Service KF6::KIOGui KF6::WindowSystem)
target_compile_definitions(kamakiri-studio PRIVATE STUDIO_BUNDLE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(kamakiri-studio PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME updater-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/updater_test.py)
 add_test(NAME installer-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/installer_test.py)
 add_executable(studio-tests tests/core_test.cpp)
 target_link_libraries(studio-tests PRIVATE studio_core Qt6::Test)
 add_test(NAME studio-tests COMMAND studio-tests)
 add_executable(popup-tests tests/popup_test.cpp src/popup.cpp)
 target_link_libraries(popup-tests PRIVATE studio_core Qt6::Test KF6::Service KF6::KIOGui KF6::WindowSystem)
 add_test(NAME popup-tests COMMAND popup-tests)
 set_tests_properties(popup-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 set_tests_properties(studio-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS kamakiri-studio RUNTIME DESTINATION bin)
install(FILES kamakiri-studio.desktop DESTINATION share/applications)
